  script={`// A small raster engine: OpenStreetMap tiles on the flat Web Mercator plane, drag to pan.
// Applications usually pass createMapLibreEngine(maplibregl) from '@tweakpad/ui/map' instead.
// Every map on this page marked data-engine="tiles" uses it.
const TILE = 256;
const toWorld = (p) => {
  const s = Math.sin((p.latitude * Math.PI) / 180);
  return { x: ((p.longitude + 180) / 360) * TILE, y: (0.5 - Math.log((1 + s) / (1 - s)) / (4 * Math.PI)) * TILE };
};
const fromWorld = (w) => ({
  longitude: (w.x / TILE) * 360 - 180,
  latitude: (Math.atan(Math.sinh(Math.PI * (1 - (2 * w.y) / TILE))) * 180) / Math.PI,
});

const osmTiles = {
  name: 'osm-raster',
  mount(viewport, context) {
    let camera = context.initialCamera;
    let interactive = context.interactive;
    const overlays = new Map();
    const images = new Map();
    const surface = document.createElement('div');
    const tiles = document.createElement('div');
    const pane = document.createElement('div');
    const credit = document.createElement('a');
    surface.style.cssText = 'position:absolute;inset:0;overflow:hidden;touch-action:none;cursor:grab;background-color:var(--tp-muted);background-image:linear-gradient(var(--tp-border) 1px,transparent 1px),linear-gradient(90deg,var(--tp-border) 1px,transparent 1px)';
    tiles.style.cssText = 'position:absolute;inset:0';
    pane.style.cssText = 'position:absolute;inset:0;pointer-events:none';
    credit.href = 'https://www.openstreetmap.org/copyright';
    credit.textContent = '© OpenStreetMap contributors';
    credit.style.cssText = 'position:absolute;inset-block-end:0;inset-inline-end:0;padding:2px 6px;font-size:var(--tp-text-xs);color:var(--tp-muted-foreground);background:var(--tp-background)';
    surface.append(tiles);
    viewport.append(surface, pane, credit);

    const size = () => ({ width: viewport.clientWidth, height: viewport.clientHeight });
    const project = (p) => {
      const c = toWorld(camera.center), w = toWorld(p), k = 2 ** camera.zoom, s = size();
      return { x: s.width / 2 + (w.x - c.x) * k, y: s.height / 2 + (w.y - c.y) * k };
    };
    const unproject = (pt) => {
      const c = toWorld(camera.center), k = 2 ** camera.zoom, s = size();
      return fromWorld({ x: c.x + (pt.x - s.width / 2) / k, y: c.y + (pt.y - s.height / 2) / k });
    };
    const draw = () => {
      const { width, height } = size();
      const z = Math.max(0, Math.min(19, Math.round(camera.zoom)));
      const n = 2 ** z, side = TILE * 2 ** (camera.zoom - z);
      const c = toWorld(camera.center), k = 2 ** camera.zoom;
      const left = c.x * k - width / 2, top = c.y * k - height / 2;
      surface.style.backgroundSize = side / 4 + 'px ' + side / 4 + 'px';
      surface.style.backgroundPosition = -left + 'px ' + -top + 'px';
      const used = new Set();
      for (let y = Math.max(0, Math.floor(top / side)); y <= Math.min(n - 1, Math.floor((top + height) / side)); y++) {
        for (let x = Math.floor(left / side); x <= Math.floor((left + width) / side); x++) {
          const key = z + '/' + (((x % n) + n) % n) + '/' + y;
          let img = images.get(key);
          if (!img) {
            img = Object.assign(new Image(), { alt: '', draggable: false, src: 'https://tile.openstreetmap.org/' + key + '.png' });
            img.style.cssText = 'position:absolute;left:0;top:0;user-select:none';
            img.onerror = () => (img.style.visibility = 'hidden');
            images.set(key, img);
            tiles.append(img);
          }
          used.add(key);
          img.style.width = img.style.height = side + 0.5 + 'px';
          img.style.transform = 'translate(' + (x * side - left) + 'px,' + (y * side - top) + 'px)';
        }
      }
      for (const [key, img] of images) if (!used.has(key)) (img.remove(), images.delete(key));
      for (const [element, position] of overlays) {
        const p = project(position);
        element.style.transform = 'translate(' + p.x + 'px,' + p.y + 'px)';
      }
    };

    let drag = null;
    surface.addEventListener('pointerdown', (event) => {
      if (!interactive) return;
      drag = { x: event.clientX, y: event.clientY, center: toWorld(camera.center), moved: false };
      surface.setPointerCapture(event.pointerId);
    });
    surface.addEventListener('pointermove', (event) => {
      if (!drag) return;
      const dx = event.clientX - drag.x, dy = event.clientY - drag.y;
      if (!drag.moved && Math.hypot(dx, dy) < 3) return;
      drag.moved = true;
      const k = 2 ** camera.zoom;
      const y = Math.min(TILE, Math.max(0, drag.center.y - dy / k));
      camera = { ...camera, center: fromWorld({ x: drag.center.x - dx / k, y }) };
      draw();
      context.camera('engine', event);
    });
    surface.addEventListener('pointerup', (event) => {
      if (!drag) return;
      if (drag.moved) context.cameraEnd('engine', event);
      else {
        const box = surface.getBoundingClientRect();
        const point = { x: event.clientX - box.left, y: event.clientY - box.top };
        context.press(unproject(point), point, event);
      }
      drag = null;
    });

    const setAppearance = (scheme) => {
      tiles.style.filter = scheme === 'dark' ? 'invert(1) hue-rotate(180deg) brightness(0.85) contrast(0.9)' : '';
    };
    setAppearance(context.scheme);
    draw();
    queueMicrotask(() => context.ready());

    return {
      native: surface,
      getCamera: () => camera,
      getBounds() {
        const s = size(), nw = unproject({ x: 0, y: 0 }), se = unproject({ x: s.width, y: s.height });
        return { south: se.latitude, west: nw.longitude, north: nw.latitude, east: se.longitude };
      },
      jumpTo(next) {
        camera = next;
        draw();
      },
      project,
      unproject,
      attachOverlay(element, position) {
        element.style.cssText += ';position:absolute;left:0;top:0;pointer-events:auto';
        overlays.set(element, position);
        pane.append(element);
        draw();
        return {
          setPosition(next) {
            overlays.set(element, next);
            draw();
          },
          remove() {
            overlays.delete(element);
            element.remove();
          },
        };
      },
      setInteractive(enabled) {
        interactive = enabled;
      },
      setAppearance,
      resize: draw,
      destroy() {
        surface.remove();
        pane.remove();
        credit.remove();
      },
    };
  },
};

for (const map of document.querySelectorAll('tp-map[data-engine="tiles"]')) map.engine = osmTiles;`}

## Usage

The map owns the camera, pins, selection, overlays and controls; the basemap comes from an engine you assign to `engine`. In an application that is `createMapLibreEngine(maplibregl)` or `createGoogleMapsEngine(importLibrary)` from `@tweakpad/ui/map`. This site ships no map library, so its examples use the small OpenStreetMap raster engine from the first example, written against the same public `MapEngine` contract.

The library has no engine dependency. Adapters in `@tweakpad/ui/map` receive the engine namespace you import, and every adapter produces the same public behavior.

Implement `MapEngine` (`mount` returning `getCamera`, `getBounds`, `jumpTo`, `project`, `unproject`, `attachOverlay` and `destroy`, plus optional animation, fit, zoom-range, interaction, appearance and resize capabilities) to add another engine.

### Pins and overlays

A pin's default slot replaces the marker with your own markup; SVG using `currentColor` follows the pin's `color`. Selecting a pin opens its `tp-map-overlay`, a non-modal Popover that follows the pin while the camera moves and closes on Escape or an outside press. `tp-map-overlay` is a non-modal Popover anchored to its pin: every Popover slot (`title`, `description`, `header`, `close`) and placement property applies (`side` defaults to `block-start`, `side-offset` to 8).

Without authored content a pin renders the default marker. Pins form one tab stop: arrow keys move between them, Enter or Space selects and Escape clears the selection. Each pin is a button with `aria-pressed` reflecting selection.

### Location list

When pins carry essential information, pair the map with a list of the same places bound to the selection. `subscribe` keeps the list in step with the map, and `selectPin` moves the map from the list. The viewport is a labelled region. Every gesture has a control or key equivalent, and the map engine keeps its own keyboard pan and zoom.

### Controls

`controls` floats a joined Button group over the map, in the order you list the actions; `controls-position` and `controls-orientation` place it. `tp-map-control` with `map="id"` puts the same buttons anywhere on the page. `controls-position` picks a logical corner (`start`/`end` mirror in right-to-left); bottom corners sit above the engine attribution. Any other floating content goes in a corner slot — `slot="top-start"`, `top-end`, `bottom-start` or `bottom-end` — and unslotted children join the `controls-position` corner after the built-in buttons. Empty corners are not rendered.

The **home view** is `default-bounds`, else `default-center` with `default-zoom`, else the fitted pins, else the whole world. The initial camera is the home view and **reset** returns to it.

### Status

Until an engine is ready the map shows a status message. Replace it through the `status` slot, for example when the basemap waits for consent. Camera requests made before the engine is ready update the pending camera and apply at mount. With reduced motion (`motion-policy="reduce"` or the user preference), camera moves apply immediately.

### Theming

The basemap follows the map's computed `color-scheme` (dark styles in a dark scope). `theme` assigns colors to ten roles per scheme — `background`, `land`, `water`, `park`, `road`, `roadMajor`, `building`, `boundary`, `label`, `labelHalo` — using any CSS color, including tokens; `tokenMapTheme` maps them to the library tokens. Call `refreshAppearance()` after changing a scoped theme that the map cannot observe.
