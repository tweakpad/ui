import React, { Fragment } from 'react';
import {
  Canvas,
  Description,
  PureArgsTable,
  Source,
  Title,
  useOf,
} from '@storybook/addon-docs/blocks';
import { MDXProvider } from '@storybook/addon-docs/mdx-react-shim';
import { manifest } from './manifest.js';
import { elementApi } from '../src/docs/api.js';
import { mountPreview, previewSource } from '../src/docs/preview-runtime.js';

export const LibraryDocContext = React.createContext(null);

export function PreviewCanvas({ preview }) {
  // The library's documentation page of a component, rendered on its Docs page with Storybook's
  // own blocks: the header, the main use case (the Default story), the page's Usage
  // (`LibraryDoc`), then the generated API Reference. `Preview`, `Code` and `Keys` are the
  // elements the pages use; the website renders the same pages with its own components. Top-level
  // comments are prose to MDX, so the notes live inside the blocks.
  const host = React.useRef(null);
  React.useEffect(() => {
    const mounted = mountPreview(host.current, preview);
    return () => mounted.dispose();
  }, [preview]);
  // A scrolling use case: the preview itself is the scroll container (and the size container that
  // pinned sections fill), at the height the example asks for.
  const style = preview.viewport
    ? {
        blockSize: preview.viewport,
        overflow: 'auto',
        containerType: 'size',
        paddingInline: 'var(--tp-space-6)',
      }
    : undefined;
  return (
    <div
      className="tp-docs-example-preview"
      data-align={preview.align || 'center'}
      ref={host}
      style={style}
    />
  );
}

export function Preview(props) {
  const context = React.useContext(LibraryDocContext);
  const preview = React.useMemo(
    () => props,
    [props.code, props.script, props.viewport, props.align],
  );
  const components = React.useMemo(
    () => ({ Story: () => <PreviewCanvas preview={preview} /> }),
    [preview],
  );
  // The hero is the Default story above; the page's first preview is not shown twice.
  const hero = context?.hero;
  if (hero && hero.title === props.title && hero.code === props.code) return null;
  const source = {
    code: previewSource(props.code, props.script),
    language: props.language || 'html',
    format: false,
  };
  return (
    <figure className="tp-docs-example">
      <figcaption>{props.title}</figcaption>
      {props.codeOnly ? (
        <Canvas withToolbar={false} source={source} sourceState="shown" />
      ) : (
        <MDXProvider components={components}>
          <Canvas
            layout={props.viewport ? 'fullscreen' : 'padded'}
            withToolbar={false}
            source={source}
          />
        </MDXProvider>
      )}
    </figure>
  );
}

export function Code({ code, language, label }) {
  // A standalone sample with nothing to render: Storybook's Source block, under its label.
  return (
    <figure className="tp-docs-example">
      {label && <figcaption>{label}</figcaption>}
      <Source code={code.trim()} language={language || 'html'} format={false} />
    </figure>
  );
}

export const KEY_ALIASES = {
  up: 'arrowup',
  down: 'arrowdown',
  left: 'arrowleft',
  right: 'arrowright',
  esc: 'escape',
};

export function Keys({ keys }) {
  const names = keys
    .split('+')
    .map((key) => key.trim())
    .filter(Boolean)
    .map((key) => KEY_ALIASES[key.toLowerCase()] ?? (key.length === 1 ? key.toUpperCase() : key));
  return (
    <tp-key-hint-group>
      {names.map((name, index) => (
        <KeyHint key={index} name={name} />
      ))}
    </tp-key-hint-group>
  );
}

export function KeyHint({ name }) {
  // `key` is reserved by React, so the attribute is set on the element itself.
  const set = React.useCallback((element) => element?.setAttribute('key', name), [name]);
  return <tp-key-hint ref={set} />;
}

export function useLibraryDoc() {
  const resolved = useOf('meta', ['meta']);
  return resolved.preparedMeta.parameters.docs?.library ?? null;
}

export function PageHeader() {
  // Name, description and tag, from the page's frontmatter. A story file without a page (an
  // internal Storybook page) keeps Storybook's own title and description.
  const doc = useLibraryDoc();
  if (!doc)
    return (
      <>
        <Title />
        <Description />
      </>
    );
  const { page } = doc;
  return (
    <>
      <Title>{page.title}</Title>
      <Description />
      <p className="tp-docs-tag">
        <code>&lt;{page.tag}&gt;</code>
      </p>
    </>
  );
}

export function LibraryDoc() {
  const doc = useLibraryDoc();
  if (!doc) return null;
  const { page, Content } = doc;
  const components = { Preview, Code, Keys };
  // MDX resolves member-expression elements through its provider, so the context provider is
  // created directly.
  return React.createElement(
    LibraryDocContext.Provider,
    { value: { hero: page.hero } },
    <MDXProvider components={components}>
      <Content components={components} />
    </MDXProvider>,
  );
}

export function ApiRows({ label, id, level, rows }) {
  // One group of API rows in Storybook's own args table (the look of the Controls table above),
  // read-only: no args, so no Control column. Descriptions are Markdown, so code spans render.
  if (!rows.length) return null;
  const Heading = level;
  const argTypes = Object.fromEntries(
    rows.map((row) => [
      row.name,
      {
        name: row.name,
        description: row.description,
        table: {
          ...(row.type ? { type: { summary: row.type } } : {}),
          ...(row.default ? { defaultValue: { summary: row.default } } : {}),
        },
      },
    ]),
  );
  return (
    <>
      <Heading id={id}>{label}</Heading>
      <PureArgsTable rows={argTypes} sort="none" />
    </>
  );
}

export function ElementApi({ tag, level }) {
  // Attributes, Properties and Events, then Methods, Slots, CSS parts and CSS custom properties,
  // each only when the element has rows for it.
  const api = elementApi(manifest, tag);
  if (!api)
    return (
      <p>
        No manifest entry for <code>{tag}</code>.
      </p>
    );
  const group = (name) => ({
    label: name,
    id: `api-${tag}-${name.toLowerCase().replace(/\s+/g, '-')}`,
    level,
  });
  const described = (text, note) => [text, note].filter(Boolean).join('\n\n');
  return (
    <>
      <ApiRows
        {...group('Attributes')}
        rows={api.attributes.map((row) => ({
          ...row,
          description: described(
            row.description,
            row.property && row.property !== row.name && `Property: \`${row.property}\``,
          ),
        }))}
      />
      <ApiRows
        {...group('Properties')}
        rows={api.properties.map((row) => ({
          ...row,
          description: described(row.description, row.readonly && 'Read-only.'),
        }))}
      />
      <ApiRows {...group('Events')} rows={api.events} />
      <ApiRows
        {...group('Methods')}
        rows={api.methods.map((row) => ({ ...row, name: row.signature, type: row.returns }))}
      />
      <ApiRows
        {...group('Slots')}
        rows={api.slots.map((row) => ({ ...row, name: row.name || 'default' }))}
      />
      <ApiRows
        {...group('CSS parts')}
        rows={api.cssParts.map((row) => ({ ...row, name: `::part(${row.name})` }))}
      />
      <ApiRows {...group('CSS custom properties')} rows={api.cssProperties} />
    </>
  );
}

export function Api() {
  const doc = useLibraryDoc();
  if (!doc) return null;
  const { page } = doc;
  return (
    <>
      <h2 id="api-reference">API Reference</h2>
      <p>
        Every element also accepts <code>motion-policy</code>.
      </p>
      <ElementApi tag={page.tag} level="h3" />
      {page.parts.map((part) => (
        <Fragment key={part}>
          <h3 id={`api-${part}`}>
            <code>&lt;{part}&gt;</code>
          </h3>
          <ElementApi tag={part} level="h4" />
        </Fragment>
      ))}
    </>
  );
}
