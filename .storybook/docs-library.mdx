export const LibraryDocContext = React.createContext(null);

export function PreviewCanvas({ preview }) {
  // The library's documentation page of a component, rendered on its Docs page with Storybook's
  // own blocks: the header, the main use case (the Default story), the page's Usage
  // (`LibraryDoc`), then the generated API Reference. `Preview`, `Code` and `Keys` are the
  // elements the pages use; the website renders the same pages with its own components. Top-level
  // comments are prose to MDX, so the notes live inside the blocks.
  const host = React.useRef(null);
  React.useEffect(() => {
    const mounted = mountPreview(host.current, preview);
    return () => mounted.dispose();
  }, [preview]);
  // A scrolling use case: the preview itself is the scroll container (and the size container that
  // pinned sections fill), at the height the example asks for.
  const style = preview.viewport
    ? {
        blockSize: preview.viewport,
        overflow: 'auto',
        containerType: 'size',
        paddingInline: 'var(--tp-space-6)',
      }
    : undefined;
  return (
    <div
      className={`tp-docs-example-preview${preview.width ? ` demo-frame demo-w-${preview.width}` : ''}`}
      data-align={preview.align || 'center'}
      ref={host}
      style={style}
    />
  );
}

export function Preview(props) {
  const context = React.useContext(LibraryDocContext);
  const preview = React.useMemo(
    () => props,
    [props.code, props.script, props.viewport, props.align, props.width],
  );
  const components = React.useMemo(
    () => ({ Story: () => <PreviewCanvas preview={preview} /> }),
    [preview],
  );
  // The hero is the Default story above; the page's first preview is not shown twice.
  const hero = context?.hero;
  if (hero && hero.title === props.title && hero.code === props.code) return null;
  const source = {
    code: previewSource(props.code, props.script),
    language: props.language || 'html',
    format: false,
  };
  // The section heading names a lone preview; a caption tells apart several in one section.
  const parsed = context?.previews.find(
    (candidate) => candidate.code === props.code && candidate.title === props.title,
  );
  const siblings = parsed
    ? context.previews.filter((candidate) => candidate.afterHeading === parsed.afterHeading).length
    : 1;
  const captioned = Boolean(props.title) && siblings > 1;
  // A page without stories (a pattern) has no story for Canvas to attach to: the preview renders
  // in its own frame, with its source below.
  if (context?.standalone)
    return (
      <figure className="tp-docs-example">
        {captioned && <figcaption>{props.title}</figcaption>}
        {!props.codeOnly && (
          <div className="tp-docs-standalone-canvas sb-unstyled">
            <PreviewCanvas preview={preview} />
          </div>
        )}
        <Source code={source.code} language={source.language} format={false} dark />
      </figure>
    );
  return (
    <figure className="tp-docs-example">
      {captioned && <figcaption>{props.title}</figcaption>}
      {props.codeOnly ? (
        <Canvas withToolbar={false} source={source} sourceState="shown" />
      ) : (
        <MDXProvider components={components}>
          <Canvas
            layout={props.viewport ? 'fullscreen' : 'padded'}
            withToolbar={false}
            source={source}
          />
        </MDXProvider>
      )}
    </figure>
  );
}

export function Code({ code, language, label }) {
  // A standalone sample with nothing to render: Storybook's Source block, under its label.
  return (
    <figure className="tp-docs-example">
      {label && <figcaption>{label}</figcaption>}
      <Source code={code.trim()} language={language || 'html'} format={false} dark />
    </figure>
  );
}

export const KEY_ALIASES = {
  up: 'arrowup',
  down: 'arrowdown',
  left: 'arrowleft',
  right: 'arrowright',
  esc: 'escape',
};

export function Kbd({ name }) {
  return <tp-kbd value={name} />;
}

export function Keys({ keys }) {
  const names = keys
    .split('+')
    .map((key) => key.trim())
    .filter(Boolean)
    .map((key) => KEY_ALIASES[key.toLowerCase()] ?? (key.length === 1 ? key.toUpperCase() : key));
  return (
    <tp-kbd-group>
      {names.map((name, index) => (
        <Kbd key={index} name={name} />
      ))}
    </tp-kbd-group>
  );
}

export function useLibraryDoc() {
  const resolved = useOf('meta', ['meta']);
  return resolved.preparedMeta.parameters.docs?.library ?? null;
}

export function PageHeader() {
  // Name, description and tag, from the page's frontmatter. A story file without a page (an
  // internal Storybook page) keeps Storybook's own title and description.
  const doc = useLibraryDoc();
  if (!doc)
    return (
      <>
        <Title />
        <Description />
      </>
    );
  const { page } = doc;
  return (
    <>
      <Title>{page.title}</Title>
      <Description />
      <p className="tp-docs-tag">
        <code>&lt;{page.tag}&gt;</code>
      </p>
    </>
  );
}

export function DocContent({ Content, hero, previews, standalone = false }) {
  // A page's body with the library's elements: component pages skip their hero (the Default
  // story shows it), pattern pages show every preview.
  const components = { Preview, Code, Keys, table: DocsTable };
  // MDX resolves member-expression elements through its provider, so the context provider is
  // created directly.
  return React.createElement(
    LibraryDocContext.Provider,
    { value: { hero, previews, standalone } },
    <MDXProvider components={components}>
      <Content components={components} />
    </MDXProvider>,
  );
}

export function LibraryDoc() {
  const doc = useLibraryDoc();
  if (!doc) return null;
  const { page, Content } = doc;
  return <DocContent Content={Content} hero={page.hero} previews={page.previews} />;
}

export function ApiRows({ label, id, level, rows, tag }) {
  // One group of API rows in Storybook's own args table (the look of the Controls table above),
  // read-only: no args, so no Control column. Descriptions are Markdown, so code spans render. A
  // row's `signature` closes its description as clickable type tokens, so the type chip is left
  // empty and the type shows once.
  if (!rows.length) return null;
  const Heading = level;
  const argTypes = Object.fromEntries(
    rows.map((row) => [
      row.name,
      {
        name: row.name,
        description: [
          rowProseHtml(tag, row.description),
          row.signature ? signatureHtml(tag, row.signature) : '',
        ]
          .filter(Boolean)
          .join('\n\n'),
        table: {
          ...(row.type && !row.signature ? { type: { summary: row.type } } : {}),
          ...(row.default ? { defaultValue: { summary: row.default } } : {}),
        },
      },
    ]),
  );
  return (
    <>
      <Heading id={id}>{label}</Heading>
      {/* A group without defaults (events, methods, slots, parts) drops the Default column, and one
      without types (slots, parts) the type placeholder under each description. */}
      <div
        className={[
          rows.some((row) => row.default) ? '' : 'tp-docs-api-no-default',
          rows.some((row) => row.type || row.signature) ? '' : 'tp-docs-api-no-type',
        ]
          .filter(Boolean)
          .join(' ')}
      >
        <PureArgsTable rows={argTypes} sort="none" />
      </div>
    </>
  );
}

export function ElementApi({ tag, level, properties = false }) {
  // What the page's Properties table does not cover: Events, Methods, Slots, CSS parts and CSS
  // custom properties, each only when the element has rows for it. A part tag has no table of its
  // own above, so it also gets its Properties table here (`properties`).
  const api = elementApi(manifest, tag);
  if (!api)
    return (
      <p>
        No manifest entry for <code>{tag}</code>.
      </p>
    );
  const group = (name) => ({
    label: name,
    id: `api-${tag}-${name.toLowerCase().replace(/\s+/g, '-')}`,
    level,
    tag,
  });
  return (
    <>
      {properties ? (
        <ApiRows
          {...group('Properties')}
          rows={api.members.map((row) => ({
            name: row.name,
            type: row.type,
            default: row.default,
            description: memberDescription(row, tag),
          }))}
        />
      ) : null}
      {api.formControl ? (
        <p>
          As a form control it also accepts <code>name</code>, <code>disabled</code>,{' '}
          <code>required</code> and <code>read-only</code>.
        </p>
      ) : null}
      <ApiRows
        {...group('Events')}
        rows={api.events.map((row) => ({ ...row, signature: row.type }))}
      />
      <ApiRows
        {...group('Methods')}
        rows={api.methods.map((row) => ({
          ...row,
          signature: `${row.signature}: ${row.returns || 'void'}`,
        }))}
      />
      <ApiRows
        {...group('Slots')}
        rows={api.slots.map((row) => ({ ...row, name: row.name || 'default' }))}
      />
      <ApiRows
        {...group('CSS parts')}
        rows={api.cssParts.map((row) => ({ ...row, name: `::part(${row.name})` }))}
      />
      <ApiRows {...group('CSS custom properties')} rows={api.cssProperties} />
    </>
  );
}

export function CardTokens({ lines }) {
  // Highlighted lines of a declaration; a library type name is a button that opens its card (the
  // page's TypeCard handles the click).
  return lines.map((line, index) => (
    <Fragment key={index}>
      {index ? '\n' : null}
      {line.map((token, at) => {
        const style = token.scope ? { color: `var(--tp-syntax-${token.scope})` } : undefined;
        return token.ref ? (
          <button
            key={at}
            type="button"
            className="tp-docs-type-ref"
            style={style}
            data-tp-ref={token.ref}
          >
            {token.text}
          </button>
        ) : (
          <span key={at} style={style}>
            {token.text}
          </span>
        );
      })}
    </Fragment>
  ));
}

export function TypeCard() {
  // The page's one type card, an editor hover on click: a type name in a signature opens the
  // card of its type at the name (declaration, doc comment, import line), and the names inside
  // the card open theirs, with a back step. A native popover: light dismiss (Escape, outside
  // press) and no library elements in the docs chrome.
  const popover = React.useRef(null);
  const [stack, setStack] = React.useState([]);
  React.useEffect(() => {
    const open = (event) => {
      const button = event.target instanceof Element ? event.target.closest('[data-tp-ref]') : null;
      const id = button?.getAttribute('data-tp-ref');
      if (!id || !symbolCard(types, id)) return;
      // A name inside the card (its declaration or doc comment) opens the next card in place.
      if (popover.current?.contains(button)) {
        setStack((entries) => [...entries, { id, anchor: entries[0].anchor }]);
        return;
      }
      const rect = button.getBoundingClientRect();
      setStack([{ id, anchor: { left: rect.left, top: rect.bottom, above: rect.top } }]);
    };
    document.addEventListener('click', open);
    return () => document.removeEventListener('click', open);
  }, []);
  const shown = stack.at(-1);
  const card = shown ? symbolCard(types, shown.id) : undefined;
  React.useLayoutEffect(() => {
    const element = popover.current;
    if (!element || !card) return;
    if (!element.matches(':popover-open')) element.showPopover();
    // At the name that opened it, inside the viewport: below it, or above when there is no room.
    const { left, top, above } = stack[0].anchor;
    const gap = 6;
    const margin = 16;
    const { width, height } = element.getBoundingClientRect();
    element.style.left = `${Math.max(margin, Math.min(left, innerWidth - width - margin))}px`;
    element.style.top = `${
      top + gap + height <= innerHeight - margin || above - gap - height < margin
        ? Math.max(margin, Math.min(top + gap, innerHeight - height - margin))
        : above - gap - height
    }px`;
  }, [card, stack]);
  return (
    <div
      ref={popover}
      popover="auto"
      className="tp-docs-type-card"
      onToggle={(event) => {
        if (event.newState === 'closed') setStack([]);
      }}
    >
      {card ? (
        <>
          <header>
            {stack.length > 1 ? (
              <button
                type="button"
                className="tp-docs-type-back"
                onClick={() => setStack((entries) => entries.slice(0, -1))}
                aria-label={`Back to ${symbolCard(types, stack.at(-2).id)?.name ?? 'previous type'}`}
              >
                ←
              </button>
            ) : null}
            <code>{card.name}</code>
            <span className="tp-docs-type-kind">{card.kind}</span>
          </header>
          <pre>
            <CardTokens lines={card.lines} />
          </pre>
          {card.documentation ? (
            <div className="tp-docs-type-doc">
              <Markdown>
                {proseHtml(card.documentation, typeRefs(types, { symbol: card.id }))}
              </Markdown>
            </div>
          ) : null}
          {card.importLine ? (
            <footer>
              <code>{card.importLine}</code>
            </footer>
          ) : card.platform ? (
            <footer>
              <span className="tp-docs-type-kind">Built into the platform: no import needed.</span>
            </footer>
          ) : null}
        </>
      ) : null}
    </div>
  );
}

export function Api() {
  const doc = useLibraryDoc();
  if (!doc) return null;
  const { page } = doc;
  return (
    <>
      <h2 id="api-reference">API Reference</h2>
      <p>
        Every element also accepts <code>motion-policy</code>.
      </p>
      <ElementApi tag={page.tag} level="h3" />
      {page.parts.map((part) => (
        <Fragment key={part}>
          <h3 id={`api-${part}`}>
            <code>&lt;{part}&gt;</code>
          </h3>
          <ElementApi tag={part} level="h4" properties />
        </Fragment>
      ))}
    </>
  );
}
